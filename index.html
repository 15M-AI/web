[ COMMAND LINE INTERFACE ]

Setup & Code Integration

Official snippets for cloning and using 15M repositories.

bash - 15m-cli
[100% OPEN SOURCE]
# 1. Clone the official repository from GitHub
git clone https://github.com/15M-AI/pico.git
# 2. Load model weights directly via Hugging Face Transformers
from transformers import AutoModelForCausalLM, AutoTokenizer

# Load tokenizer & weights from Hugging Face
tokenizer = AutoTokenizer.from_pretrained("15MAI/pico")
model = AutoModelForCausalLM.from_pretrained("15MAI/pico")

inputs = tokenizer("Hello Pico, generate a response", return_tensors="pt")
outputs = model.generate(**inputs, max_new_tokens=100)
print(tokenizer.decode(outputs[0]))
